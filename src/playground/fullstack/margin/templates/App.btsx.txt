import { useEffect, useRef, useState } from 'octane'
import { useField, useForm } from '@octanejs/tanstack-form'
import BookCover from './BookCover.btsx'
import { excerptFor } from './excerpts'
import { categories, setSaved, checkout, loadProducts, loadSession, money, setQuantity, type CartItem, type Order, type Product } from './store'

setup
  const [products, setProducts] = useState<Product[]>([]);
  const [cart, setCart] = useState<CartItem[]>([]);
  const [order, setOrder] = useState<Order | null>(null);
  const [shelf, setShelf] = useState<string[]>([]);
  const [onlySaved, setOnlySaved] = useState(false);
  const [sample, setSample] = useState(false);
  const [category, setCategory] = useState('All');
  const [selected, setSelected] = useState<Product | null>(null);
  const [loading, setLoading] = useState(true);
  const [pending, setPending] = useState(false);
  const [error, setError] = useState('');
  const [notice, setNotice] = useState('');
  const [motion, setMotion] = useState(true);
  const [checkoutKey, setCheckoutKey] = useState(() => crypto.randomUUID());
  const details = useRef<HTMLDialogElement | null>(null);
  const bag = useRef<HTMLDialogElement | null>(null);
  const form = useForm({ defaultValues: { search: '', sort: 'featured', budget: 60, delivery: 'standard' } });
  const search = useField({ form, name: 'search' });
  const sort = useField({ form, name: 'sort' });
  const budget = useField({ form, name: 'budget' });
  const delivery = useField({ form, name: 'delivery' });
  const query = search.state.value.trim().toLowerCase();
  const visible = products.filter(product => (!onlySaved || shelf.includes(product.sku)) && (category === 'All' || product.category === category) && product.price <= budget.state.value * 100 && (product.name + ' ' + product.description + ' ' + product.category + ' ' + product.sku).toLowerCase().includes(query)).sort((a, b) => sort.state.value === 'price-low' ? a.price - b.price : sort.state.value === 'price-high' ? b.price - a.price : sort.state.value === 'name' ? a.name.localeCompare(b.name) : Number(Boolean(b.badge)) - Number(Boolean(a.badge)) || a.sku.localeCompare(b.sku));
  const lines = cart.flatMap(item => { const product = products.find(product => product.sku === item.sku); return product ? [{ ...item, product }] : []; });
  const count = cart.reduce((total, item) => total + item.quantity, 0);
  const subtotal = lines.reduce((total, item) => total + item.product.price * item.quantity, 0);
  const shipping = delivery.state.value === 'express' ? 1200 : subtotal >= 7500 ? 0 : 500;
  const load = async (signal?: AbortSignal) => {
    try {
      const [catalog, session] = await Promise.all([loadProducts(signal), loadSession(signal)]);
      if (signal?.aborted) return;
      setShelf(session.shelf); setProducts(catalog); setCart(session.items); setOrder(session.order); setError(''); setLoading(false);
    } catch (reason) {
      if (!signal?.aborted) { setError(reason instanceof Error ? reason.message : 'Unable to open the store.'); setLoading(false); }
    }
  };
  useEffect(() => {
    const abort = new AbortController();
    void load(abort.signal);
    return () => { abort.abort(); details.current?.close(); bag.current?.close(); };
  }, []);
  const changeQuantity = async (product: Product, quantity: number) => {
    if (pending) return;
    setPending(true); setError('');
    try {
      const items = await setQuantity(product.sku, quantity);
      setCart(items); setNotice(quantity ? product.name + ' is in your bag.' : product.name + ' removed.');
      setCheckoutKey(crypto.randomUUID());
    } catch (reason) { setError(reason instanceof Error ? reason.message : 'Could not update your bag.'); }
    finally { setPending(false); }
  };
  const add = (product: Product) => void changeQuantity(product, (cart.find(item => item.sku === product.sku)?.quantity ?? 0) + 1);
  const openProduct = (product: Product) => { setSample(false); setSelected(product); details.current?.showModal(); };
  const openBag = () => { details.current?.close(); bag.current?.showModal(); };
  const placeOrder = async () => {
    if (pending || !cart.length) return;
    setPending(true); setError('');
    try {
      const receipt = await checkout(checkoutKey, delivery.state.value === 'express' ? 'express' : 'standard');
      setOrder(receipt); setCart([]); setNotice('Demo order ' + receipt.reference + ' is confirmed. No payment was taken.');
    } catch (reason) { setError(reason instanceof Error ? reason.message : 'Checkout failed. Your bag is still here.'); }
    finally { setPending(false); }
  };
  const resetFilters = () => { setOnlySaved(false); setCategory('All'); search.handleChange(''); sort.handleChange('featured'); budget.handleChange(60); };

  const saveBook = async (product: Product) => {
    if (pending) return;
    setPending(true); setError('');
    try { const skus = await setSaved(product.sku, !shelf.includes(product.sku)); setShelf(skus); setNotice(skus.includes(product.sku) ? 'A place on your shelf, saved.' : 'Removed from your reading shelf.'); }
    catch (reason) { setError(reason instanceof Error ? reason.message : 'Could not save this book.'); }
    finally { setPending(false); }
  };
  const showShelf = () => { setOnlySaved(true); setCategory('All'); search.handleChange(''); budget.handleChange(60); document.querySelector('#collection')?.scrollIntoView({ behavior: 'smooth' }); };

    else
      div.book-grid
